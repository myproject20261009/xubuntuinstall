import subprocess
import sys

def run_json_lsblk():
    result = subprocess.run(
        ["lsblk", "--json", "--bytes", "--nodeps",

        check=True, text=True, capture_output=True
    )

    disks = []
    for item in run_json_lsblk():

            continue
        if str(item.get("ro", "0")) == "1":

        path = item.get("path")

            disks.append(item)
    return disks

    # JSON double-quoted strings are valid YAML strings.
    return json.dumps(str(value), ensure_ascii=False)

        sys.exit("Confirmation did not match; no configuration generated.")

    partitions = [

        emit(f"  number: {number}")
        emit(f"  size: {yaml_quote(size)}")
        if flag:

    formats = [
        ("efi-format", "efi-part", "fat32", "EFI"),

        emit("  type: format")

    for crypt_id, volume, name, keyfile in crypts:
        emit(f"- id: {crypt_id}")
        emit("  type: dm_crypt")
        emit(f"  volume: {volume}")
        emit(f"  dm_name: {name}")

        ("root-mount", "root-format", "/"),
        ("boot-mount", "boot-format", "/boot"),

        emit(f"- id: {mount_id}")
        emit("  type: mount")
        emit(f"  device: {device}")
        emit(f"  path: {path}")

    print("\n# Unallocated space remains unpartitioned.")
    print("# Inspect this YAML; this generator does not execute it.")
