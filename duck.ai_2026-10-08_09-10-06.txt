#!/usr/bin/env python3
"""Review-only Curtin YAML generator. Does not partition, format, or install."""

import json
import os
import subprocess
import sys

GIB = 1024 ** 3
MIB = 1024 ** 2

def run_json_lsblk():
    result = subprocess.run(
        ["lsblk", "--json", "--bytes", "--nodeps",
         "-o", "PATH,SIZE,MODEL,TRAN,TYPE,RO"],
        check=True, text=True, capture_output=True
    )
    return json.loads(result.stdout).get("blockdevices", [])

def find_readwrite_disks():
    disks = []
    for item in run_json_lsblk():
        if item.get("type") != "disk":
            continue
        if str(item.get("ro", "0")) == "1":
            continue
        path = item.get("path")
        size = item.get("size")
        if path and isinstance(size, int):
            disks.append(item)
    return disks

def show_keyfile_mount(path):
    result = subprocess.run(
        ["findmnt", "-n", "-o", "SOURCE,FSTYPE,TARGET", "-T", path],
        check=True, text=True, capture_output=True
    )
    return result.stdout.strip()

def yaml_quote(value):
    # JSON double-quoted strings are valid YAML strings.
    return json.dumps(str(value), ensure_ascii=False)

def main():
    disks = find_readwrite_disks()
    if not disks:
        sys.exit("No writable whole disks found; no configuration generated.")

    print("Detected whole disks (this script will not modify them):")
    for i, disk in enumerate(disks, 1):
        model = (disk.get("model") or "").strip()
        transport = (disk.get("tran") or "").strip()
        print(
            f"  {i}. {disk['path']}  "
            f"{disk['size'] / GIB:.1f} GiB  "
            f"{model}  {transport}"
        )

    try:
        choice = int(input("Select target disk number: ").strip())
        if not 1 <= choice <= len(disks):
            raise ValueError
    except ValueError:
        sys.exit("Invalid selection; no configuration generated.")

    disk = disks[choice - 1]
    disk_path = disk["path"]
    print(f"\nSelected target: {disk_path}")
    print("This selection will appear in a YAML file containing wipe actions.")
    if input(f'Type exactly "USE {disk_path}" to continue: ').strip() != f"USE {disk_path}":
        sys.exit("Confirmation did not match; no configuration generated.")

    try:
        home_gib = int(input("Desired /home size in whole GiB: ").strip())
        if home_gib <= 0:
            raise ValueError
    except ValueError:
        sys.exit("Enter a positive whole number of GiB; no configuration generated.")

    # EFI 1 GiB + /boot 2 GiB + root 45 GiB + chosen /home,
    # plus a small allowance for GPT/alignment.
    required = (1 + 2 + 45 + home_gib) * GIB + 32 * MIB
    if required > disk["size"]:
        sys.exit(
            "Requested layout does not fit this disk; "
            "no configuration generated."
        )

    root_key = input("Path to root keyfile on the unlocked encrypted disk: ").strip()
    home_key = input("Path to home keyfile on the unlocked encrypted disk: ").strip()

    for label, path in (("root", root_key), ("home", home_key)):

        "\nConfirm: both keyfiles are on an unlocked encrypted disk "
        "different from the selected target disk."
    )
    if input('Type exactly "KEYFILES ON SEPARATE ENCRYPTED DISK": ').strip() != \
            "KEYFILES ON SEPARATE ENCRYPTED DISK":
        sys.exit("Confirmation did not match; no configuration generated.")

    print("\n# Review-only generated storage configuration.")
    print("# This output is not an install command.")
    print("storage:")

        emit("  wipe: superblock")

"""Read-only Curtin storage YAML generator.

This script only reads block-device and mount information, asks questions,
and prints YAML to stdout. It does not run Curtin or modify disks/files.
"""

import json
import os
import subprocess
import sys

GIB = 1024 ** 3
MIB = 1024 ** 2


def fail(message):
    sys.exit(message)


def run_json_lsblk():
    result = subprocess.run(
        [
            "lsblk", "--json", "--bytes",
            "--output", "PATH,SIZE,MODEL,TRAN,TYPE,RO,MOUNTPOINTS",
        ],
        check=True,
        text=True,
        capture_output=True,
    )
    return json.loads(result.stdout).get("blockdevices", [])


def has_mounted_descendant(device):
    """Return True if this device or any child has a mountpoint."""
    mountpoints = device.get("mountpoints") or []
    if isinstance(mountpoints, str):
        mountpoints = [mountpoints]

    if any(point for point in mountpoints):
        return True

    return any(
        has_mounted_descendant(child)
        for child in (device.get("children") or [])
    )


def find_readwrite_unmounted_disks():
    disks = []

    for item in run_json_lsblk():
        if item.get("type") != "disk":
            continue

        # lsblk may represent RO as an integer or string.
        if str(item.get("ro", "0")) == "1":
            continue

        path = item.get("path")
        size = item.get("size")
        if not path or not isinstance(size, int):
            continue

        # Avoid offering disks with mounted filesystems or mounted children.
        if has_mounted_descendant(item):
            continue

        disks.append(item)

    return disks


def yaml_quote(value):
    # JSON double-quoted strings are valid YAML strings.
    return json.dumps(str(value), ensure_ascii=False)


def readable_regular_file(path):
    return (
        bool(path)
        and os.path.isfile(path)
        and os.access(path, os.R_OK)
    )

        print("    " + line)

    print("# Do not pass this YAML to Curtin until it has been fully reviewed.")
    print("storage:")
    print("  version: 1")
    print("  config:")

    emit("- id: selected-disk")
    emit("  type: disk")
    emit(f"  path: {yaml_quote(disk_path)}")
    emit("  ptable: gpt")
    emit("  wipe: superblock")
    emit("  grub_device: true")

        ("efi-part", 1, "1G", "boot"),
        ("boot-part", 2, "2G", None),
        ("root-part", 3, "45G", None),
        ("home-part", 4, f"{home_gib}G", None),

        emit(f"- id: {part_id}")
        emit("  type: partition")
        emit("  device: selected-disk")
        emit(f"  number: {number}")
        emit(f"  size: {yaml_quote(size)}")
        if flag:
            emit(f"  flag: {flag}")
        emit("  wipe: superblock")

        emit("  type: mount")
        emit(f"  device: {device}")
        emit(f"  path: {path}")

    print()
    print("# No swap action is configured.")
    print("# No partition is defined for the remaining disk space.")
